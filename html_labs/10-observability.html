Lab 16: Observe Agent Runs and Build a Metrics Dashboard

Operational teams need more than final answers. Traces, latency, token usage, and success status reveal where an agent spends time and where failures occur.

Why this matters

Operational teams need more than final answers. Traces, latency, token usage, and success status reveal where an agent spends time and where failures occur.

Learning objectives

  • Configure OpenTelemetry providers
  • Create and read an agent trace
  • Generate a self-contained HTML metrics dashboard from real run telemetry

Core concepts

ConceptWhat it means in this lab
TraceConnects spans from one end-to-end operation.
SpanMeasures one operation such as a model or tool call.
Operational metricsSummarize success, latency, tokens, and span counts.

Create or verify required resources

  2. The shared key is for classroom model inference only. Never commit, print, screenshot, or submit it.

async def main() -> None:
    # Console exporters keep the first observability demo self-contained.
    configure_otel_providers(enable_console_exporters=True)

    agent = Agent(
        client=OpenAIChatClient(
            base_url=os.environ["AZURE_OPENAI_ENDPOINT"],
            api_key=os.environ["AZURE_OPENAI_API_KEY"],
            model=os.environ["AZURE_OPENAI_DEPLOYMENT"],
        ),
        name="ObservedAgent",
        instructions="Use check_service_health for status questions. Keep answers brief.",
        tools=[check_service_health],
    )

    with get_tracer().start_as_current_span("classroom-agent-demo", kind=SpanKind.CLIENT) as span:
        print(f"Trace ID: {format_trace_id(span.get_span_context().trace_id)}")
        result = await agent.run(
            "Check the payment service and explain in one sentence why tracing helps support teams."
        )
        print(result.text)

def token_count(usage: dict, name: str) -> int:
    value = usage.get(name, 0)
    return value if isinstance(value, int) else 0

    const telemetry = __TELEMETRY_DATA__;
    const runs = Array.isArray(telemetry.runs) ? telemetry.runs : [];
    const spans = Array.isArray(telemetry.spans) ? telemetry.spans : [];

    const number = new Intl.NumberFormat("en-US");
    const successful = runs.filter((run) => run.success).length;
    const totalLatency = runs.reduce((sum, run) => sum + Number(run.latency_ms || 0), 0);
    const totalTokens = runs.reduce((sum, run) => sum + Number(run.total_tokens || 0), 0);
    const averageLatency = runs.length ? totalLatency / runs.length : 0;

    document.getElementById("run-count").textContent = number.format(runs.length);
    document.getElementById("success-rate").textContent = runs.length
      ? `${Math.round((successful / runs.length) * 100)}%`
      : "0%";
    document.getElementById("average-latency").textContent = `${number.format(Math.round(averageLatency))} ms`;
    document.getElementById("total-tokens").textContent = number.format(totalTokens);
    document.getElementById("span-count").textContent = number.format(spans.length);
    document.getElementById("service-name").textContent = `Service: ${telemetry.service_name || "agent-framework"}`;
    document.getElementById("generated-at").textContent = `Generated: ${new Date(telemetry.generated_at).toLocaleString()}`;
    document.getElementById("source-label").textContent = telemetry.source || "Local telemetry";

    function createBarChart(containerId, values, valueLabel, stacked = false) {
      const container = document.getElementById(containerId);
      if (!values.length) {
        container.innerHTML = '<div class="empty">No run data was recorded.</div>';
        return;
      }

      const max = Math.max(1, ...values.map((item) => stacked
        ? Number(item.input_tokens || 0) + Number(item.output_tokens || 0)
        : Number(item.latency_ms || 0)));

      values.forEach((item) => {
        const row = document.createElement("div");
        row.className = "chart-row";
        const label = document.createElement("div");
        label.className = "chart-label";
        const name = document.createElement("span");
        name.textContent = item.label;
        const value = document.createElement("span");
        value.textContent = valueLabel(item);
        label.append(name, value);

        const track = document.createElement("div");
        track.className = "track";
        if (stacked) {
          const input = document.createElement("div");
          input.className = "bar input";
          input.style.width = `${(Number(item.input_tokens || 0) / max) * 100}%`;
          const output = document.createElement("div");
          output.className = "bar output";
          output.style.width = `${(Number(item.output_tokens || 0) / max) * 100}%`;
          track.append(input, output);
        } else {
          const bar = document.createElement("div");
          bar.className = "bar";
          bar.style.width = `${(Number(item.latency_ms || 0) / max) * 100}%`;
          track.append(bar);
        }
        row.append(label, track);
        container.append(row);
      });
    }

    createBarChart("latency-chart", runs, (run) => `${number.format(Math.round(run.latency_ms || 0))} ms`);
    createBarChart(
      "token-chart",
      runs,
      (run) => number.format(Number(run.input_tokens || 0) + Number(run.output_tokens || 0)),
      true,
    );

    function renderRunsTable() {
      const container = document.getElementById("runs-table");
      if (!runs.length) {
        container.innerHTML = '<div class="empty">No runs were attempted.</div>';
        return;
      }
      const table = document.createElement("table");
      table.innerHTML = "<thead><tr><th>Run</th><th>Status</th><th class='number'>Latency</th><th class='number'>Tokens</th></tr></thead>";
      const body = document.createElement("tbody");
      runs.forEach((run) => {
        const row = document.createElement("tr");
        const name = document.createElement("td");
        name.textContent = run.label;
        const status = document.createElement("td");
        const pill = document.createElement("span");
        pill.className = run.success ? "pill" : "pill failed";
        pill.textContent = run.success ? "Succeeded" : (run.error_type || "Failed");
        status.append(pill);
        const latency = document.createElement("td");
        latency.className = "number";
        latency.textContent = `${number.format(Math.round(run.latency_ms || 0))} ms`;
        const tokens = document.createElement("td");
        tokens.className = "number";
        tokens.textContent = number.format(run.total_tokens || 0);
        row.append(name, status, latency, tokens);
        body.append(row);
      });
      table.append(body);
      container.append(table);
    }

    function renderSpansTable() {
      const container = document.getElementById("spans-table");
      if (!spans.length) {
        container.innerHTML = '<div class="empty">No spans were exported.</div>';
        return;
      }
      const table = document.createElement("table");
      table.innerHTML = "<thead><tr><th>Span</th><th>Status</th><th class='number'>Duration</th></tr></thead>";
      const body = document.createElement("tbody");
      [...spans]
        .sort((a, b) => Number(b.duration_ms || 0) - Number(a.duration_ms || 0))
        .slice(0, 10)
        .forEach((span) => {
          const row = document.createElement("tr");
          const name = document.createElement("td");
          name.textContent = span.name;
          name.title = `Trace ${span.trace_id} / Span ${span.span_id}`;
          const status = document.createElement("td");
          status.textContent = span.status;
          const duration = document.createElement("td");
          duration.className = "number";
          duration.textContent = `${number.format(Math.round(span.duration_ms || 0))} ms`;
          row.append(name, status, duration);
          body.append(row);
        });
      table.append(body);
      container.append(table);
    }

    function createBarChart(containerId, values, valueLabel, stacked = false) {
      const container = document.getElementById(containerId);
      if (!values.length) {
        container.innerHTML = '<div class="empty">No run data was recorded.</div>';
        return;
      }

      const max = Math.max(1, ...values.map((item) => stacked
        ? Number(item.input_tokens || 0) + Number(item.output_tokens || 0)
        : Number(item.latency_ms || 0)));

      values.forEach((item) => {
        const row = document.createElement("div");
        row.className = "chart-row";
        const label = document.createElement("div");
        label.className = "chart-label";
        const name = document.createElement("span");
        name.textContent = item.label;
        const value = document.createElement("span");
        value.textContent = valueLabel(item);
        label.append(name, value);

        const track = document.createElement("div");
        track.className = "track";
        if (stacked) {
          const input = document.createElement("div");
          input.className = "bar input";
          input.style.width = `${(Number(item.input_tokens || 0) / max) * 100}%`;
          const output = document.createElement("div");
          output.className = "bar output";
          output.style.width = `${(Number(item.output_tokens || 0) / max) * 100}%`;
          track.append(input, output);
        } else {
          const bar = document.createElement("div");
          bar.className = "bar";
          bar.style.width = `${(Number(item.latency_ms || 0) / max) * 100}%`;
          track.append(bar);
        }
        row.append(label, track);
        container.append(row);
      });
    }

    function renderRunsTable() {
      const container = document.getElementById("runs-table");
      if (!runs.length) {
        container.innerHTML = '<div class="empty">No runs were attempted.</div>';
        return;
      }
      const table = document.createElement("table");
      table.innerHTML = "<thead><tr><th>Run</th><th>Status</th><th class='number'>Latency</th><th class='number'>Tokens</th></tr></thead>";
      const body = document.createElement("tbody");
      runs.forEach((run) => {
        const row = document.createElement("tr");
        const name = document.createElement("td");
        name.textContent = run.label;
        const status = document.createElement("td");
        const pill = document.createElement("span");
        pill.className = run.success ? "pill" : "pill failed";
        pill.textContent = run.success ? "Succeeded" : (run.error_type || "Failed");
        status.append(pill);
        const latency = document.createElement("td");
        latency.className = "number";
        latency.textContent = `${number.format(Math.round(run.latency_ms || 0))} ms`;
        const tokens = document.createElement("td");
        tokens.className = "number";
        tokens.textContent = number.format(run.total_tokens || 0);
        row.append(name, status, latency, tokens);
        body.append(row);
      });
      table.append(body);
      container.append(table);
    }
